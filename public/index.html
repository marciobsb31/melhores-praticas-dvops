📜 Código do Jenkinsfile (Pipeline Declarativa com Aprovação)

// Pipeline Declarativa Jenkins - Unyleya Unidade 2
// Aluno: Márcio Pereira | Repositório: marciobsb31/melhores-praticas-dvops
pipeline {
    agent any

    environment {
        APP_NAME     = 'melhores-praticas-devops'
        STUDENT_NAME = 'Márcio Pereira'
        REPO_URL     = 'https://github.com/marciobsb31/melhores-praticas-dvops'
        SONAR_HOST   = 'http://localhost:9000'
        DEV_PORT     = '3000'
        PRD_PORT     = '3001'
    }

    stages {
        stage('1. Checkout SCM') {
            steps {
                echo "Clonando repositório GitHub: ${env.REPO_URL}"
                git branch: 'main', url: "${env.REPO_URL}"
            }
        }

        stage('2. Build & Dependências') {
            steps {
                echo "Instalando dependências do projeto..."
                sh 'npm ci'
            }
        }

        stage('3. Testes Automatizados (CI)') {
            steps {
                echo "Executando suíte de testes unitários com Jest..."
                sh 'npm run test:coverage'
            }
        }

        stage('4. Análise de Qualidade SonarQube') {
            steps {
                echo "Iniciando SonarQube Scanner para validação de Quality Gate..."
                withSonarQubeEnv('SonarQubeServer') {
                    sh 'sonar-scanner -Dsonar.projectKey=melhores-praticas-dvops -Dsonar.projectName="melhores-praticas-dvops - Marcio Pereira"'
                }
                timeout(time: 5, unit: 'MINUTES') {
                    script {
                        def qg = waitForQualityGate()
                        if (qg.status != 'OK') {
                            error "Quality Gate falhou no SonarQube: ${qg.status}"
                        }
                    }
                }
            }
        }

        stage('5. Deploy em Desenvolvimento (DEV)') {
            steps {
                echo "Publicando aplicação no Ambiente Local de Desenvolvimento (Porta ${env.DEV_PORT})..."
                sh 'NODE_ENV=development PORT=3000 npm run start &'
                echo "Ambiente DEV publicado com sucesso em http://localhost:3000"
            }
        }

        stage('6. Solicitação de Liberação para Produção') {
            steps {
                echo "Aguardando aprovação manual do Engenheiro DevOps para deploy em Produção..."
                input message: 'Deseja autorizar a publicação da versão no ambiente de PRODUÇÃO?', 
                      ok: 'Liberar Deploy em Produção', 
                      submitter: 'marcio'
            }
        }

        stage('7. Deploy em Produção (PRD)') {
            steps {
                echo "Publicando aplicação no Ambiente Local de Produção (Porta ${env.PRD_PORT})..."
                sh 'NODE_ENV=production PORT=3001 npm run start &'
                echo "Ambiente PRD publicado com sucesso em http://localhost:3001"
            }
        }
    }

    post {
        success {
            echo "Pipeline executada com SUCESSO total por Márcio Pereira!"
        }
        failure {
            echo "Falha detectada durante a execução da pipeline CI/CD."
        }
    }
}
